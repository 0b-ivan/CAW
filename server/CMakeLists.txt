# CAW backend and firmware rules regression suite.
cmake_minimum_required(VERSION 3.24)
project(caw VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CAW_BUILD_SERVER "Build the HTTP server and database clients" ON)
option(CAW_DEPENDENCIES_ONLY "Prepare dependencies without application sources" OFF)
include(CTest)
include(FetchContent)
find_package(doctest 2.4.9 CONFIG QUIET)
if(NOT doctest_FOUND)
  FetchContent_Declare(doctest GIT_REPOSITORY https://github.com/doctest/doctest.git
    GIT_TAG v2.4.11 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(doctest)
endif()
add_library(caw_warnings INTERFACE)
target_compile_options(caw_warnings INTERFACE -Wall -Wextra -Werror -funsigned-char)
add_library(caw_game INTERFACE)
target_include_directories(caw_game INTERFACE src/game)
include(cmake/Dependencies.cmake)
if(CAW_DEPENDENCIES_ONLY)
  return()
endif()
add_library(caw_support src/config.cpp src/log.cpp src/net/Metrics.cpp)
target_include_directories(caw_support PUBLIC src)
target_link_libraries(caw_support PRIVATE caw_warnings)
add_library(caw_session src/session/Codes.cpp src/session/SessionRegistry.cpp)
target_include_directories(caw_session PUBLIC src)
target_link_libraries(caw_session PRIVATE caw_warnings)
add_library(caw_messages src/net/Messages.cpp)
target_include_directories(caw_messages PUBLIC src)
target_link_libraries(caw_messages PUBLIC caw_game nlohmann_json::nlohmann_json PRIVATE caw_warnings)
if(CAW_BUILD_SERVER)
  add_executable(caw-server src/main.cpp src/store/Health.cpp)
  target_include_directories(caw-server PRIVATE src)
  target_link_libraries(caw-server PRIVATE caw_support caw_session caw_messages caw_warnings
    uwebsockets PkgConfig::PQXX ${CAW_REDIS_TARGET})
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
